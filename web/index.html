        # Minimal Astris server: hands out the 26.3 folder and its file list.
import json
from functools import partial
from http.server import SimpleHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

ROOT = Path("26.3")                    # the prepared folder (with horizon-resources.json)
ORIGIN = "https://astris.example"      # where the launcher is opened
PORT = 8790

LISTING = json.dumps([[p.relative_to(ROOT).as_posix(), p.stat().st_size]
                      for p in sorted(ROOT.rglob("*")) if p.is_file()]).encode()


class Source(SimpleHTTPRequestHandler):
    def end_headers(self):
        self.send_header("Access-Control-Allow-Origin", ORIGIN)
        self.send_header("Cross-Origin-Resource-Policy", "cross-origin")
        super().end_headers()

    def do_GET(self):
        if self.path == "/__astris.json":
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(LISTING)))
            self.end_headers()
            self.wfile.write(LISTING)
        else:
            super().do_GET()

    def list_directory(self, path):
        self.send_error(403)           # files only, no folder listings

    def log_message(self, *args):
        pass


ThreadingHTTPServer(("0.0.0.0", PORT), partial(Source, directory=str(ROOT))).serve_forever()
